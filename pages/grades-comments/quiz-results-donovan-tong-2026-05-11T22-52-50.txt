Question 7:
Add defensive checks so this function safely handles missing model or title data.

Answer:
type LessonCard = { title?: string };

function getTitleUpper(card?: LessonCard): string {
  // return "UNTITLED" when card or title is missing
  if (!card || card.title === "") {
     return "UNTITLED";
  }
  return card.title.toUpperCase();
}

Question 12:
Write a generic function getFirst<T> that returns the first array item, or undefined for an empty array.

Answer:
function getFirst<T>(items: T[]): T | undefined {
  if (items.length > 0) {
    return items[0];
  }
  return undefined;
}
